التعديل 3: تغيير JavaScript شريط الكوكيز (حفظ 3 أيام)
====================================================================
الموقع: السطر 17156 — 17200
الوصف: إضافة localStorage بحيث بعد ما المستخدم يضغط "قبول"
        ما يظهرش تاني إلا بعد 3 أيام
السبب: تحسين تجربة المستخدم

الـ JS القديم (للرجوع):
    (function(){
        'use strict';
        var cookieTriggered = false;

        function showCookieBar() {
            if (cookieTriggered) return;
            var bar = document.getElementById('cookieConsent');
            if (bar) {
                bar.classList.add('show');
                cookieTriggered = true;
                
                var acceptBtn = document.getElementById('acceptCookies');
                var rejectBtn = document.getElementById('rejectCookies');
                if (acceptBtn) acceptBtn.onclick = function(e) { e.preventDefault(); hideCookieBar(); };
                if (rejectBtn) rejectBtn.onclick = function(e) { e.preventDefault(); hideCookieBar(); };
            }
        }

        function hideCookieBar() {
            var bar = document.getElementById('cookieConsent');
            if (bar) {
                bar.classList.remove('show');
            }
        }

        window.addEventListener('load', function() {
            setTimeout(showCookieBar, 8000); 
        });

        var userInteractions = ['mouseover', 'keydown', 'touchstart', 'scroll'];
        userInteractions.forEach(function(event) {
            window.addEventListener(event, showCookieBar, {once: true, passive: true});
        });

    })();

الـ JS الجديد (الموجود حالياً):
    (function(){
        'use strict';
        var cookieTriggered = false;
        var COOKIE_KEY = 'seoturbo_cookie_consent';
        var THREE_DAYS = 3 * 24 * 60 * 60 * 1000;

        function shouldShowBar() {
            var saved = localStorage.getItem(COOKIE_KEY);
            if (!saved) return true;
            var ts = parseInt(saved, 10);
            if (isNaN(ts)) return true;
            return (Date.now() - ts) >= THREE_DAYS;
        }

        function showCookieBar() {
            if (cookieTriggered) return;
            if (!shouldShowBar()) return;
            var bar = document.getElementById('cookieConsent');
            if (bar) {
                bar.classList.add('show');
                cookieTriggered = true;
                
                var acceptBtn = document.getElementById('acceptCookies');
                var rejectBtn = document.getElementById('rejectCookies');
                if (acceptBtn) acceptBtn.onclick = function(e) { 
                    e.preventDefault(); 
                    localStorage.setItem(COOKIE_KEY, Date.now().toString());
                    hideCookieBar(); 
                };
                if (rejectBtn) rejectBtn.onclick = function(e) { 
                    e.preventDefault(); 
                    hideCookieBar(); 
                };
            }
        }

        function hideCookieBar() {
            var bar = document.getElementById('cookieConsent');
            if (bar) {
                bar.classList.remove('show');
            }
        }

        window.addEventListener('load', function() {
            setTimeout(showCookieBar, 8000); 
        });

        var userInteractions = ['mouseover', 'keydown', 'touchstart', 'scroll'];
        userInteractions.forEach(function(event) {
            window.addEventListener(event, showCookieBar, {once: true, passive: true});
        });

    })();
